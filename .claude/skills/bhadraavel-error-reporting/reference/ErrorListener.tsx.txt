"use client";

import { useEffect } from "react";

// Hears what the error page can't: a throw in an event handler, a promise
// nobody awaited, a failed chunk load after a deploy. Posts to /api/errors,
// which adds the signed-in user and hands it to the support hub. Never
// throws, never blocks, and the page's own URL goes as a template-ish path
// (ids replaced) rather than the real one.
export function routeOf(path: string): string {
  return path.replace(/\/\d+(?=\/|$)/g, "/[id]").replace(/\/(a|d)\/[^/]+/g, "/$1/[token]").replace(/\?.*$/, "");
}

export function reportToServer(e: { name?: string; message?: string; stack?: string; digest?: string; kind: string }) {
  try {
    const payload = JSON.stringify({ ...e, route: routeOf(location.pathname), ua: navigator.userAgent, viewport: `${innerWidth}x${innerHeight}` });
    if (navigator.sendBeacon) navigator.sendBeacon("/api/errors", new Blob([payload], { type: "application/json" }));
    else fetch("/api/errors", { method: "POST", headers: { "content-type": "application/json" }, body: payload, keepalive: true }).catch(() => {});
  } catch { /* never */ }
}

const NOISE = /ResizeObserver loop|Script error\.?$|chrome-extension:|moz-extension:|safari-extension:/;

export default function ErrorListener() {
  useEffect(() => {
    const onError = (ev: ErrorEvent) => {
      if (NOISE.test(ev.message ?? "") || (ev.filename && /extension:/.test(ev.filename))) return;
      const err = ev.error as Error | undefined;
      reportToServer({ name: err?.name ?? "Error", message: err?.message ?? ev.message, stack: err?.stack, kind: "window.onerror" });
    };
    const onRejection = (ev: PromiseRejectionEvent) => {
      const r = ev.reason as Error | string | undefined;
      const message = typeof r === "string" ? r : r?.message ?? "Unhandled promise rejection";
      if (NOISE.test(message) || (typeof r === "object" && r?.name === "AbortError")) return;
      reportToServer({ name: typeof r === "object" ? r?.name ?? "Error" : "Error", message, stack: typeof r === "object" ? r?.stack : undefined, kind: "unhandledrejection" });
    };
    window.addEventListener("error", onError);
    window.addEventListener("unhandledrejection", onRejection);
    return () => { window.removeEventListener("error", onError); window.removeEventListener("unhandledrejection", onRejection); };
  }, []);
  return null;
}
